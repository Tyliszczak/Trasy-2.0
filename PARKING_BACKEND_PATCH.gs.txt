/*
 * TRASY 2.0 — administracyjne zapisywanie Bazy/Parkingu
 *
 * Funkcję wolno wywołać wyłącznie PO istniejącej, poprawnej autoryzacji
 * administratora. Nie umieszczaj tego warunku przed sprawdzeniem hasła/tokena.
 *
 * W autoryzowanej części doPost(e) dodaj:
 *
 *   if (action === 'upsertParking') {
 *     return upsertParking_(postData);
 *   }
 *
 * Następnie utwórz nowe wdrożenie Apps Script. Główny doGet powinien zwracać
 * kartę PARKINGI razem z pozostałymi danymi aplikacji.
 */

function upsertParking_(p) {
  var name = String(p.name || '').trim();
  var route = String(p.route || '*').trim() || '*';
  var match = String(p.coordinates || '').match(/^\s*(-?\d+(?:\.\d+)?)\s*[,; ]\s*(-?\d+(?:\.\d+)?)\s*$/);

  if (!name || name.length > 80 || /^[=+@-]/.test(name)) {
    return createJsonResponse({status:'error', message:'Nieprawidłowa nazwa Bazy/Parkingu.'});
  }
  if (route.length > 120 || /^[=+@-]/.test(route)) {
    return createJsonResponse({status:'error', message:'Nieprawidłowa nazwa trasy.'});
  }
  if (!match) {
    return createJsonResponse({status:'error', message:'Nieprawidłowe współrzędne.'});
  }

  var lat = Number(match[1]), lng = Number(match[2]);
  if (!isFinite(lat) || !isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180) {
    return createJsonResponse({status:'error', message:'Współrzędne są poza dozwolonym zakresem.'});
  }

  var coordinates = lat + ', ' + lng;
  var lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = spreadsheet.getSheetByName('PARKINGI');
    if (!sheet) {
      sheet = spreadsheet.insertSheet('PARKINGI');
      sheet.getRange(1, 1, 1, 3).setValues([['NAZWA', 'LOKALIZACJA', 'TRASA']]);
      sheet.setFrozenRows(1);
    }

    var values = sheet.getDataRange().getValues();
    var targetRow = 0;
    for (var row = 1; row < values.length; row++) {
      if (String(values[row][0] || '').trim().toLowerCase() === name.toLowerCase() &&
          String(values[row][2] || '*').trim().toLowerCase() === route.toLowerCase()) {
        targetRow = row + 1;
        break;
      }
    }
    if (!targetRow) targetRow = Math.max(2, sheet.getLastRow() + 1);
    sheet.getRange(targetRow, 1, 1, 3).setValues([[name, coordinates, route]]);
    SpreadsheetApp.flush();
    return createJsonResponse({status:'success', parking:{name:name, coordinates:coordinates, route:route}});
  } catch (error) {
    return createJsonResponse({status:'error', message:String(error)});
  } finally {
    lock.releaseLock();
  }
}
