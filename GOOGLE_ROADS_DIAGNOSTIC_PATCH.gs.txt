/*
 * TRASY 2.0 — DIAGNOSTYKA GOOGLE ROADS / SPEED LIMITS
 *
 * TEST, NIE IMPLEMENTACJA SPEEDMAX.
 * Nie zwraca ani nie loguje klucza API.
 * Używa istniejącej właściwości Apps Script: GOOGLE_MAPS_API_KEY.
 *
 * Do doPost(e), zaraz po odczytaniu action/postData, dodaj:
 *
 *   if (action === 'diagnoseGoogleRoads') {
 *     return diagnoseGoogleRoads_(postData);
 *   }
 *
 * Po wdrożeniu test wysyła:
 *   {"action":"diagnoseGoogleRoads","latitude":51.961460,"longitude":15.499002}
 */

function diagnoseGoogleRoads_(p) {
  var props = PropertiesService.getScriptProperties();
  var apiKey = props.getProperty('GOOGLE_MAPS_API_KEY') || '';

  if (!apiKey) {
    return createJsonResponse({
      status: 'error',
      stage: 'key',
      code: 'GOOGLE_KEY_MISSING',
      message: 'Brak GOOGLE_MAPS_API_KEY we właściwościach Apps Script.'
    });
  }

  var lat = Number(p && p.latitude);
  var lng = Number(p && p.longitude);
  if (!isFinite(lat) || !isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180) {
    return createJsonResponse({
      status: 'error',
      stage: 'input',
      code: 'INVALID_COORDINATES',
      message: 'Nieprawidłowe współrzędne testowe.'
    });
  }

  function requestJson_(url) {
    var resp = UrlFetchApp.fetch(url, {
      method: 'get',
      muteHttpExceptions: true,
      headers: { Accept: 'application/json' }
    });
    var http = resp.getResponseCode();
    var text = resp.getContentText();
    var data = {};
    try { data = JSON.parse(text || '{}'); } catch (_) {}
    return { http: http, data: data };
  }

  function googleError_(stage, result) {
    var error = result && result.data && result.data.error;
    return createJsonResponse({
      status: 'error',
      stage: stage,
      http: result ? result.http : 0,
      code: error && error.status ? error.status : ('HTTP_' + (result ? result.http : 0)),
      message: error && error.message ? String(error.message) : 'Google Roads API zwróciło błąd.'
    });
  }

  try {
    var nearestUrl = 'https://roads.googleapis.com/v1/nearestRoads' +
      '?points=' + encodeURIComponent(lat + ',' + lng) +
      '&key=' + encodeURIComponent(apiKey);

    var nearest = requestJson_(nearestUrl);
    if (nearest.http < 200 || nearest.http >= 300) {
      return googleError_('nearestRoads', nearest);
    }

    var snapped = nearest.data && nearest.data.snappedPoints;
    if (!Array.isArray(snapped) || !snapped.length || !snapped[0].placeId) {
      return createJsonResponse({
        status: 'error',
        stage: 'nearestRoads',
        http: nearest.http,
        code: 'NO_SNAPPED_POINT',
        message: 'nearestRoads działa, ale nie zwróciło placeId dla punktu testowego.'
      });
    }

    var placeId = String(snapped[0].placeId);
    var speedUrl = 'https://roads.googleapis.com/v1/speedLimits' +
      '?placeId=' + encodeURIComponent(placeId) +
      '&units=KPH' +
      '&key=' + encodeURIComponent(apiKey);

    var speed = requestJson_(speedUrl);
    if (speed.http < 200 || speed.http >= 300) {
      return googleError_('speedLimits', speed);
    }

    var limits = speed.data && speed.data.speedLimits;
    if (!Array.isArray(limits) || !limits.length) {
      return createJsonResponse({
        status: 'error',
        stage: 'speedLimits',
        http: speed.http,
        code: 'NO_SPEED_LIMIT',
        message: 'Speed Limits odpowiedziało poprawnie, ale nie zwróciło limitu dla odcinka testowego.',
        nearestRoads: true
      });
    }

    return createJsonResponse({
      status: 'success',
      provider: 'google-roads-speed-limits',
      nearestRoads: true,
      speedLimits: true,
      speedLimit: Number(limits[0].speedLimit),
      units: String(limits[0].units || 'KPH')
    });

  } catch (err) {
    return createJsonResponse({
      status: 'error',
      stage: 'exception',
      code: 'GOOGLE_ROADS_EXCEPTION',
      message: String(err && err.message ? err.message : err)
    });
  }
}
