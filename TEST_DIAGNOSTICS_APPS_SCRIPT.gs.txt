/*
 * TRASY 2.0 TEST — odbiór diagnostyki z Cloudflare.
 *
 * 1. Dodaj ten plik do projektu Apps Script przypisanego do arkusza „Trasy 2.0”.
 * 2. W Script Properties ustaw DIAGNOSTICS_SHARED_SECRET (ta sama wartość co
 *    sekret Cloudflare Pages) oraz DIAGNOSTICS_FOLDER_ID (ID prywatnego folderu
 *    na Dysku Google przeznaczonego na pliki diagnostyczne).
 * 3. W doPost(e), zaraz po utworzeniu scriptProperties, dodaj:
 *
 *    if (action === 'appendTestDiagnostics') {
 *      return appendTestDiagnostics_(postData);
 *    }
 *
 * 4. Utwórz nową wersję istniejącego wdrożenia aplikacji internetowej.
 *
 * Każda sesja diagnostyczna jest zapisywana w jednym pliku JSON. Kolejne
 * paczki są scalane z tym samym plikiem; backend nie tworzy już części 2, 3 itd.
 */

function appendTestDiagnostics_(p) {
  var properties = PropertiesService.getScriptProperties();
  var expected = String(properties.getProperty('DIAGNOSTICS_SHARED_SECRET') || '');
  var folderId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') ||
    properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  var received = String(p && p.secret || '');
  if (!expected || !constantTimeEqualsDiagnostics_(expected, received)) {
    return createJsonResponse({status:'error', message:'UNAUTHORIZED'});
  }
  if (!/^[A-Za-z0-9_-]{10,180}$/.test(folderId)) {
    return createJsonResponse({status:'error', message:'STORAGE_NOT_CONFIGURED'});
  }

  var batchId = safeDiagnosticsText_(p.batchId, 180);
  var installationId = safeDiagnosticsText_(p.installationId, 100);
  var deviceLabel = safeDiagnosticsText_(p.deviceLabel, 80);
  var appVersion = safeDiagnosticsText_(p.appVersion, 24);
  var sessionId = safeDiagnosticsText_(p.sessionId, 180);
  var events = Array.isArray(p.events) ? p.events : [];
  if (!/^[A-Za-z0-9:._-]+$/.test(batchId) ||
      !/^[A-Za-z0-9:._-]+$/.test(installationId) ||
      /[\u0000-\u001F\u007F]/.test(deviceLabel) ||
      !/^2\.0\.\d+$/.test(appVersion) ||
      !/^[A-Za-z0-9:._-]+$/.test(sessionId) ||
      !events.length || events.length > 500) {
    return createJsonResponse({status:'error', message:'INVALID_PAYLOAD'});
  }

  var normalized = [];
  var previousEventId = 0;
  for (var i = 0; i < events.length; i++) {
    var event = events[i] || {};
    if (String(event.sessionId || '') !== sessionId ||
        !isFinite(Number(event.id)) ||
        Number(event.id) <= previousEventId ||
        !/^\d{4}-\d{2}-\d{2}T/.test(String(event.at || '')) ||
        !/^[A-Za-z0-9:._-]+$/.test(String(event.type || ''))) {
      return createJsonResponse({status:'error', message:'INVALID_EVENT'});
    }
    previousEventId = Number(event.id);
    normalized.push(event);
  }

  var json = JSON.stringify(normalized);
  if (events.length > 500 || json.length > 480000) {
    return createJsonResponse({status:'error', message:'PAYLOAD_TOO_LARGE'});
  }

  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  var diagnosticFile = null;
  var sessionResult = null;
  try {
    var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = spreadsheet.getSheetByName('DIAGNOSTYKA TESTÓW');
    if (!sheet) sheet = createTestDiagnosticsSheet_(spreadsheet);
    sheet.getRange(1, 17).setValue('ZAKRES_ID');
    sheet.getRange(1, 18).setValue('PLIK_JSON');
    sheet.getRange(1, 19).setValue('URZĄDZENIE');

    var folder = getDiagnosticsJsonRoot_(properties);
    var deviceFolder = getDiagnosticsDeviceFolder_(folder, properties, installationId, deviceLabel);
    var seenRanges = getDiagnosticsSeenRanges_(properties, sheet, installationId, sessionId);
    var newEvents = [];
    for (var u = 0; u < normalized.length; u++) {
      if (!isDiagnosticsEventSeen_(seenRanges, Number(normalized[u].id))) newEvents.push(normalized[u]);
    }
    if (!newEvents.length) {
      return createJsonResponse({
        status:'success', batchId:batchId, duplicate:true,
        acceptedEvents:0, duplicateEvents:normalized.length
      });
    }

    sessionResult = upsertDiagnosticsSessionFile_(
      deviceFolder, properties, installationId, deviceLabel, appVersion, sessionId, newEvents
    );
    diagnosticFile = sessionResult.file;
    var archive = sessionResult.archive;
    var first = archive.events[0];
    var last = archive.events[archive.events.length - 1];
    var snapshot = last.snapshot || first.snapshot || {};
    var counts = summarizeDiagnosticsEvents_(archive.events);
    var rowNumber = findDiagnosticsSessionRow_(sheet, sessionResult.indexKey);
    var row = [[
      safeSheetValue_(sessionResult.indexKey), safeSheetValue_(sessionId), safeSheetValue_(installationId),
      safeSheetValue_(archive.appVersions.join(', ')), new Date(), safeSheetValue_(first.at), safeSheetValue_(last.at),
      safeSheetValue_(snapshot.route || ''), safeSheetValue_(snapshot.shift || ''),
      safeSheetValue_(snapshot.direction || ''), archive.events.length, counts.gps, counts.skipped,
      counts.errors, safeSheetValue_(first.at), safeSheetValue_(last.at), safeSheetValue_(eventRangesText_(archive.events)),
      safeSheetValue_(diagnosticFile.getUrl()), safeSheetValue_(deviceLabel)
    ]];
    sheet.getRange(rowNumber, 1, 1, 19).setValues(row);
    SpreadsheetApp.flush();
    addDiagnosticsSeenEvents_(seenRanges, normalized);
    saveDiagnosticsSeenRanges_(properties, installationId, sessionId, seenRanges);
    return createJsonResponse({
      status:'success', batchId:batchId, duplicate:sessionResult.addedEvents === 0,
      acceptedEvents:sessionResult.addedEvents,
      duplicateEvents:normalized.length - sessionResult.addedEvents,
      fileId:diagnosticFile.getId()
    });
  } catch (error) {
    if (diagnosticFile && sessionResult && sessionResult.created) {
      try { diagnosticFile.setTrashed(true); } catch (cleanupError) {}
    }
    return createJsonResponse({status:'error', message:'WRITE_FAILED'});
  } finally {
    lock.releaseLock();
  }
}

function constantTimeEqualsDiagnostics_(a, b) {
  var length = Math.max(a.length, b.length), mismatch = a.length ^ b.length;
  for (var i = 0; i < length; i++) mismatch |= (a.charCodeAt(i % Math.max(1, a.length)) || 0) ^ (b.charCodeAt(i % Math.max(1, b.length)) || 0);
  return mismatch === 0;
}

function safeDiagnosticsText_(value, max) {
  return String(value == null ? '' : value).trim().slice(0, max);
}

function safeSheetValue_(value) {
  var text = String(value == null ? '' : value);
  return /^[=+@-]/.test(text) ? "'" + text : text;
}

function diagnosticsSessionHash_(installationId, sessionId) {
  var bytes = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(installationId) + '|' + String(sessionId)
  );
  var hex = '';
  for (var i = 0; i < bytes.length; i++) {
    hex += ('0' + ((bytes[i] + 256) % 256).toString(16)).slice(-2);
  }
  return hex.slice(0, 32);
}

function diagnosticsSeenKey_(installationId, sessionId) {
  return 'DIAGNOSTICS_SEEN_' + diagnosticsSessionHash_(installationId, sessionId);
}

function normalizeDiagnosticsRanges_(ranges) {
  var sorted = (ranges || []).map(function(range) {
    return [Number(range[0]), Number(range[1])];
  }).filter(function(range) {
    return isFinite(range[0]) && isFinite(range[1]) && range[0] > 0 && range[1] >= range[0];
  }).sort(function(a, b) { return a[0] - b[0]; });
  var merged = [];
  for (var i = 0; i < sorted.length; i++) {
    var last = merged[merged.length - 1];
    if (!last || sorted[i][0] > last[1] + 1) merged.push(sorted[i]);
    else last[1] = Math.max(last[1], sorted[i][1]);
  }
  return merged;
}

function parseDiagnosticsRanges_(value) {
  var ranges = [];
  String(value || '').split(',').forEach(function(part) {
    var match = part.trim().match(/^(\d+)(?:-(\d+))?$/);
    if (match) ranges.push([Number(match[1]), Number(match[2] || match[1])]);
  });
  return normalizeDiagnosticsRanges_(ranges);
}

function getDiagnosticsSeenRanges_(properties, sheet, installationId, sessionId) {
  var key = diagnosticsSeenKey_(installationId, sessionId);
  var saved = properties.getProperty(key);
  if (saved !== null) {
    try { return normalizeDiagnosticsRanges_(JSON.parse(saved)); } catch (savedError) {}
  }

  var ranges = [];
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var identities = sheet.getRange(2, 2, lastRow - 1, 2).getValues();
    var chunks = sheet.getRange(2, 17, lastRow - 1, 1).getValues();
    for (var i = 0; i < identities.length; i++) {
      if (String(identities[i][0]) === sessionId && String(identities[i][1]) === installationId) {
        ranges = ranges.concat(parseDiagnosticsRanges_(chunks[i][0]));
      }
    }
  }
  ranges = normalizeDiagnosticsRanges_(ranges);
  properties.setProperty(key, JSON.stringify(ranges));
  return ranges;
}

function isDiagnosticsEventSeen_(ranges, eventId) {
  for (var i = 0; i < ranges.length; i++) {
    if (eventId < ranges[i][0]) return false;
    if (eventId <= ranges[i][1]) return true;
  }
  return false;
}

function addDiagnosticsSeenEvents_(ranges, events) {
  for (var i = 0; i < events.length; i++) ranges.push([Number(events[i].id), Number(events[i].id)]);
  var merged = normalizeDiagnosticsRanges_(ranges);
  ranges.length = 0;
  for (var n = 0; n < merged.length; n++) ranges.push(merged[n]);
}

function saveDiagnosticsSeenRanges_(properties, installationId, sessionId, ranges) {
  properties.setProperty(diagnosticsSeenKey_(installationId, sessionId), JSON.stringify(ranges));
}

function newDiagnosticsSessionArchive_(installationId, deviceLabel, sessionId) {
  var now = new Date().toISOString();
  return {
    format: 'trasy-2.0-test-diagnostics-session',
    schemaVersion: 2,
    installationId: installationId,
    deviceLabel: deviceLabel,
    sessionId: sessionId,
    appVersions: [],
    receivedFirstAt: now,
    receivedLastAt: now,
    locationDataIncluded: true,
    events: []
  };
}

function safeDiagnosticsFilePart_(value, maxLength) {
  var safe = String(value || '').replace(/[\\\/:*?"<>|\u0000-\u001F\u007F]+/g, '-')
    .replace(/\s+/g, '-').replace(/^-+|-+$/g, '');
  return (safe || 'nieznane').slice(0, maxLength);
}

function diagnosticsFileTime_(value) {
  var date = new Date(value);
  if (!value || isNaN(date.getTime())) return 'czas-nieznany';
  var timezone = Session.getScriptTimeZone() || 'Europe/Warsaw';
  return Utilities.formatDate(date, timezone, 'yyyy-MM-dd_HH-mm-ss');
}

function diagnosticsSessionFileSuffix_(sessionHash) {
  return '-' + String(sessionHash).slice(0, 8) + '.json';
}

function diagnosticsSessionFileName_(archive, sessionHash) {
  var events = archive && Array.isArray(archive.events) ? archive.events : [];
  var first = events.length ? events[0] : {};
  var last = events.length ? events[events.length - 1] : first;
  var start = diagnosticsFileTime_(first.at || archive.receivedFirstAt);
  var end = diagnosticsFileTime_(last.at || archive.receivedLastAt);
  var device = safeDiagnosticsFilePart_(archive.deviceLabel, 55);
  var installation = safeDiagnosticsFilePart_(archive.installationId, 8);
  return 'trasy-2.0-' + start + '--' + end + '-' + device + '-' + installation +
    diagnosticsSessionFileSuffix_(sessionHash);
}

function readDiagnosticsSessionArchive_(file, fallback) {
  if (!file) return fallback;
  try {
    var parsed = JSON.parse(file.getBlob().getDataAsString('UTF-8'));
    if (parsed && parsed.format === 'trasy-2.0-test-diagnostics-session' && Array.isArray(parsed.events)) {
      return parsed;
    }
  } catch (readError) {}
  return fallback;
}

function getDiagnosticsSessionFile_(deviceFolder, properties, stateKey, sessionHash) {
  var state = {};
  try { state = JSON.parse(properties.getProperty(stateKey) || '{}') || {}; } catch (stateError) {}
  if (state.fileId) {
    try { return DriveApp.getFileById(String(state.fileId)); } catch (savedFileError) {}
  }
  var suffix = diagnosticsSessionFileSuffix_(sessionHash);
  var files = deviceFolder.getFiles();
  while (files.hasNext()) {
    var candidate = files.next();
    if (String(candidate.getName()).slice(-suffix.length) === suffix) return candidate;
  }
  return null;
}

function overwriteDiagnosticsFileContent_(fileId, serialized) {
  var response = UrlFetchApp.fetch(
    'https://www.googleapis.com/upload/drive/v3/files/' + encodeURIComponent(fileId) + '?uploadType=media',
    {
      method:'patch',
      headers:{Authorization:'Bearer ' + ScriptApp.getOAuthToken()},
      contentType:'application/json; charset=utf-8',
      payload:serialized,
      muteHttpExceptions:true
    }
  );
  var code = response.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('DRIVE_CONTENT_UPDATE_FAILED_' + code);
}

function upsertDiagnosticsSessionFile_(deviceFolder, properties, installationId, deviceLabel, appVersion, sessionId, events) {
  var sessionHash = diagnosticsSessionHash_(installationId, sessionId);
  var stateKey = 'DIAGNOSTICS_SESSION_' + sessionHash;
  var file = getDiagnosticsSessionFile_(deviceFolder, properties, stateKey, sessionHash);
  var archive = readDiagnosticsSessionArchive_(file, newDiagnosticsSessionArchive_(installationId, deviceLabel, sessionId));
  var existingIds = {};
  for (var i = 0; i < archive.events.length; i++) existingIds[String(archive.events[i].id)] = true;
  var addedEvents = 0;
  for (var n = 0; n < events.length; n++) {
    var id = String(events[n].id);
    if (!existingIds[id]) {
      archive.events.push(events[n]);
      existingIds[id] = true;
      addedEvents++;
    }
  }
  archive.events.sort(function(a, b) { return Number(a.id) - Number(b.id); });
  if (archive.appVersions.indexOf(appVersion) < 0) archive.appVersions.push(appVersion);
  archive.deviceLabel = deviceLabel || archive.deviceLabel;
  archive.receivedLastAt = new Date().toISOString();
  var serialized = JSON.stringify(archive);
  var sizeBytes = Utilities.newBlob(serialized).getBytes().length;
  var maxBytes = 45 * 1024 * 1024;
  if (sizeBytes > maxBytes) throw new Error('SESSION_FILE_TOO_LARGE');

  var created = false;
  var fileName = diagnosticsSessionFileName_(archive, sessionHash);
  if (file) {
    if (sizeBytes <= 9 * 1024 * 1024) file.setContent(serialized);
    else overwriteDiagnosticsFileContent_(file.getId(), serialized);
    file.setName(fileName);
  }
  else {
    file = deviceFolder.createFile(Utilities.newBlob(serialized, 'application/json', fileName));
    created = true;
  }
  properties.setProperty(stateKey, JSON.stringify({fileId:file.getId()}));
  return {
    file:file, archive:archive, created:created, addedEvents:addedEvents,
    indexKey:'session:' + sessionHash
  };
}

function summarizeDiagnosticsEvents_(events) {
  var result = {gps:0, skipped:0, errors:0};
  for (var i = 0; i < events.length; i++) {
    var type = String(events[i].type || '');
    if (type === 'gps-fix') result.gps++;
    if (type === 'gps-stop-skipped') result.skipped++;
    if (/error|unhandled-rejection|gps-error/.test(type)) result.errors++;
  }
  return result;
}

function eventRangesText_(events) {
  var ranges = [];
  for (var i = 0; i < events.length; i++) ranges.push([Number(events[i].id), Number(events[i].id)]);
  return normalizeDiagnosticsRanges_(ranges).map(function(range) {
    return range[0] === range[1] ? String(range[0]) : String(range[0]) + '-' + String(range[1]);
  }).join(',');
}

function findDiagnosticsSessionRow_(sheet, indexKey) {
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    var found = sheet.getRange(2, 1, lastRow - 1, 1)
      .createTextFinder(indexKey).matchEntireCell(true).findNext();
    if (found) return found.getRow();
  }
  return Math.max(2, lastRow + 1);
}

function getDiagnosticsJsonRoot_(properties) {
  var directId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') || '');
  if (directId) {
    var directFolder = DriveApp.getFolderById(directId);
    if (directFolder.getName() === 'Pliki JSON') return directFolder;
    var directJsonFolders = directFolder.getFoldersByName('Pliki JSON');
    return directJsonFolders.hasNext() ? directJsonFolders.next() : directFolder;
  }

  var appFolderId = String(properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  var appFolder = DriveApp.getFolderById(appFolderId);
  var jsonFolders = appFolder.getFoldersByName('Pliki JSON');
  return jsonFolders.hasNext() ? jsonFolders.next() : appFolder.createFolder('Pliki JSON');
}

function getDiagnosticsDeviceFolder_(rootFolder, properties, installationId, deviceLabel) {
  var propertyKey = 'DIAGNOSTICS_DEVICE_FOLDER_' + installationId;
  var savedId = String(properties.getProperty(propertyKey) || '');
  if (savedId) {
    try {
      return DriveApp.getFolderById(savedId);
    } catch (savedFolderError) {}
  }

  var shortId = installationId.slice(0, 8);
  var suffix = ' - ' + shortId;
  var folders = rootFolder.getFolders();
  while (folders.hasNext()) {
    var existing = folders.next();
    var existingName = String(existing.getName() || '');
    if (existingName.slice(-suffix.length) === suffix) {
      properties.setProperty(propertyKey, existing.getId());
      return existing;
    }
  }

  var cleanLabel = String(deviceLabel || 'Urządzenie')
    .replace(/[\\\/:*?"<>|\u0000-\u001F]/g, ' ')
    .replace(/\s+/g, ' ').trim().slice(0, 80) || 'Urządzenie';
  var created = rootFolder.createFolder(cleanLabel + suffix);
  properties.setProperty(propertyKey, created.getId());
  return created;
}

function organizeExistingDiagnostics() {
  var properties = PropertiesService.getScriptProperties();
  var folderId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') ||
    properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  if (!/^[A-Za-z0-9_-]{10,180}$/.test(folderId)) throw new Error('STORAGE_NOT_CONFIGURED');

  var labels = {
    'a1ebd3e5-e4b5-47c3-8039-290b20c23c19': 'Android Chrome 360x800',
    '887dc03e-6a82-4720-92a9-9e6e887478c1': 'iPhone Safari 430x932',
    '8795c5b1-97cb-4dda-b058-d5e64c90ad71': 'Android Chrome 384x832',
    '83592448-5fc1-452c-a22f-94ca7ff54789': 'Android Chrome 360x780'
  };
  var rootFolder = getDiagnosticsJsonRoot_(properties);
  var files = rootFolder.getFiles();
  var moved = 0;
  var skipped = 0;
  while (files.hasNext()) {
    var file = files.next();
    var match = String(file.getName() || '').match(/trasy-2\.0-\d+\.\d+\.\d+-([0-9a-f-]{36})-/i);
    if (!match) {
      skipped++;
      continue;
    }
    var installationId = match[1].toLowerCase();
    var deviceFolder = getDiagnosticsDeviceFolder_(
      rootFolder, properties, installationId, labels[installationId] || 'Urządzenie'
    );
    file.moveTo(deviceFolder);
    moved++;
  }
  return {moved:moved, skipped:skipped};
}

function createTestDiagnosticsSheet_(spreadsheet) {
  var sheet = spreadsheet.insertSheet('DIAGNOSTYKA TESTÓW');
  sheet.getRange(1, 1, 1, 19).setValues([[
    'BATCH_ID','SESSION_ID','INSTALACJA','WERSJA','WYSŁANO','START','KONIEC','TRASA','ZMIANA','KIERUNEK',
    'ZDARZENIA','GPS','POMINIĘTE','BŁĘDY','PIERWSZE_ZDARZENIE','OSTATNIE_ZDARZENIE','ZAKRES_ID','PLIK_JSON','URZĄDZENIE'
  ]]);
  sheet.setFrozenRows(1);
  return sheet;
}
