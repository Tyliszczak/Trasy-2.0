/*
 * TRASY 2.0 TEST — odbiór diagnostyki z Cloudflare.
 *
 * 1. Dodaj ten plik do projektu Apps Script przypisanego do arkusza „Trasy 2.0”.
 * 2. W Script Properties ustaw DIAGNOSTICS_SHARED_SECRET (ta sama wartość co
 *    sekret Cloudflare Pages) oraz DIAGNOSTICS_FOLDER_ID (ID prywatnego folderu
 *    na Dysku Google przeznaczonego na pliki diagnostyczne).
 * 3. W doPost(e), zaraz po utworzeniu scriptProperties, dodaj:
 *
 *    if (action === 'appendTestDiagnostics') {
 *      return appendTestDiagnostics_(postData);
 *    }
 *
 * 4. Utwórz nową wersję istniejącego wdrożenia aplikacji internetowej.
 */

function appendTestDiagnostics_(p) {
  var properties = PropertiesService.getScriptProperties();
  var expected = String(properties.getProperty('DIAGNOSTICS_SHARED_SECRET') || '');
  var folderId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') ||
    properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  var received = String(p && p.secret || '');
  if (!expected || !constantTimeEqualsDiagnostics_(expected, received)) {
    return createJsonResponse({status:'error', message:'UNAUTHORIZED'});
  }
  if (!/^[A-Za-z0-9_-]{10,180}$/.test(folderId)) {
    return createJsonResponse({status:'error', message:'STORAGE_NOT_CONFIGURED'});
  }

  var batchId = safeDiagnosticsText_(p.batchId, 180);
  var installationId = safeDiagnosticsText_(p.installationId, 100);
  var deviceLabel = safeDiagnosticsText_(p.deviceLabel, 80);
  var appVersion = safeDiagnosticsText_(p.appVersion, 24);
  var sessionId = safeDiagnosticsText_(p.sessionId, 180);
  var events = Array.isArray(p.events) ? p.events : [];
  if (!/^[A-Za-z0-9:._-]+$/.test(batchId) ||
      !/^[A-Za-z0-9:._-]+$/.test(installationId) ||
      /[\u0000-\u001F\u007F]/.test(deviceLabel) ||
      !/^2\.0\.\d+$/.test(appVersion) ||
      !/^[A-Za-z0-9:._-]+$/.test(sessionId) ||
      !events.length || events.length > 40) {
    return createJsonResponse({status:'error', message:'INVALID_PAYLOAD'});
  }

  var normalized = [];
  for (var i = 0; i < events.length; i++) {
    var event = events[i] || {};
    if (String(event.sessionId || '') !== sessionId ||
        !isFinite(Number(event.id)) ||
        !/^\d{4}-\d{2}-\d{2}T/.test(String(event.at || '')) ||
        !/^[A-Za-z0-9:._-]+$/.test(String(event.type || ''))) {
      return createJsonResponse({status:'error', message:'INVALID_EVENT'});
    }
    normalized.push(event);
  }

  var json = JSON.stringify(normalized);
  if (json.length > 50000) return createJsonResponse({status:'error', message:'PAYLOAD_TOO_LARGE'});

  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  var diagnosticFile = null;
  try {
    var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = spreadsheet.getSheetByName('DIAGNOSTYKA TESTÓW');
    if (!sheet) sheet = createTestDiagnosticsSheet_(spreadsheet);
    sheet.getRange(1, 18).setValue('PLIK_JSON');
    sheet.getRange(1, 19).setValue('URZĄDZENIE');

    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var duplicate = sheet.getRange(2, 1, lastRow - 1, 1)
        .createTextFinder(batchId).matchEntireCell(true).findNext();
      if (duplicate) return createJsonResponse({status:'success', batchId:batchId, duplicate:true});
    }

    var folder = getDiagnosticsJsonRoot_(properties);
    var deviceFolder = getDiagnosticsDeviceFolder_(folder, properties, installationId, deviceLabel);
    var archive = {
      format: 'trasy-2.0-test-diagnostics-batch',
      schemaVersion: 1,
      batchId: batchId,
      installationId: installationId,
      deviceLabel: deviceLabel,
      appVersion: appVersion,
      sessionId: sessionId,
      receivedAt: new Date().toISOString(),
      locationDataIncluded: true,
      events: normalized
    };
    var fileName = 'trasy-2.0-' + appVersion + '-' +
      batchId.replace(/[^A-Za-z0-9._-]+/g, '-').slice(0, 140) + '.json';
    diagnosticFile = deviceFolder.createFile(
      Utilities.newBlob(JSON.stringify(archive, null, 2), 'application/json', fileName)
    );

    var first = normalized[0];
    var last = normalized[normalized.length - 1];
    var snapshot = last.snapshot || first.snapshot || {};
    var gpsCount = 0, skippedCount = 0, errorCount = 0;
    for (var n = 0; n < normalized.length; n++) {
      var type = String(normalized[n].type || '');
      if (type === 'gps-fix') gpsCount++;
      if (type === 'gps-stop-skipped') skippedCount++;
      if (/error|unhandled-rejection|gps-error/.test(type)) errorCount++;
    }
    var chunk = String(first.id) + '-' + String(last.id);
    var row = [[
      safeSheetValue_(batchId), safeSheetValue_(sessionId), safeSheetValue_(installationId),
      safeSheetValue_(appVersion), new Date(), safeSheetValue_(first.at), safeSheetValue_(last.at),
      safeSheetValue_(snapshot.route || ''), safeSheetValue_(snapshot.shift || ''),
      safeSheetValue_(snapshot.direction || ''), normalized.length, gpsCount, skippedCount,
      errorCount, safeSheetValue_(first.at), safeSheetValue_(last.at), safeSheetValue_(chunk),
      safeSheetValue_(diagnosticFile.getUrl()), safeSheetValue_(deviceLabel)
    ]];
    sheet.getRange(Math.max(2, lastRow + 1), 1, 1, 19).setValues(row);
    SpreadsheetApp.flush();
    return createJsonResponse({
      status:'success', batchId:batchId, duplicate:false,
      fileId:diagnosticFile.getId()
    });
  } catch (error) {
    if (diagnosticFile) {
      try { diagnosticFile.setTrashed(true); } catch (cleanupError) {}
    }
    return createJsonResponse({status:'error', message:'WRITE_FAILED'});
  } finally {
    lock.releaseLock();
  }
}

function constantTimeEqualsDiagnostics_(a, b) {
  var length = Math.max(a.length, b.length), mismatch = a.length ^ b.length;
  for (var i = 0; i < length; i++) mismatch |= (a.charCodeAt(i % Math.max(1, a.length)) || 0) ^ (b.charCodeAt(i % Math.max(1, b.length)) || 0);
  return mismatch === 0;
}

function safeDiagnosticsText_(value, max) {
  return String(value == null ? '' : value).trim().slice(0, max);
}

function safeSheetValue_(value) {
  var text = String(value == null ? '' : value);
  return /^[=+@-]/.test(text) ? "'" + text : text;
}

function getDiagnosticsJsonRoot_(properties) {
  var directId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') || '');
  if (directId) {
    var directFolder = DriveApp.getFolderById(directId);
    if (directFolder.getName() === 'Pliki JSON') return directFolder;
    var directJsonFolders = directFolder.getFoldersByName('Pliki JSON');
    return directJsonFolders.hasNext() ? directJsonFolders.next() : directFolder;
  }

  var appFolderId = String(properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  var appFolder = DriveApp.getFolderById(appFolderId);
  var jsonFolders = appFolder.getFoldersByName('Pliki JSON');
  return jsonFolders.hasNext() ? jsonFolders.next() : appFolder.createFolder('Pliki JSON');
}

function getDiagnosticsDeviceFolder_(rootFolder, properties, installationId, deviceLabel) {
  var propertyKey = 'DIAGNOSTICS_DEVICE_FOLDER_' + installationId;
  var savedId = String(properties.getProperty(propertyKey) || '');
  if (savedId) {
    try {
      return DriveApp.getFolderById(savedId);
    } catch (savedFolderError) {}
  }

  var shortId = installationId.slice(0, 8);
  var suffix = ' - ' + shortId;
  var folders = rootFolder.getFolders();
  while (folders.hasNext()) {
    var existing = folders.next();
    var existingName = String(existing.getName() || '');
    if (existingName.slice(-suffix.length) === suffix) {
      properties.setProperty(propertyKey, existing.getId());
      return existing;
    }
  }

  var cleanLabel = String(deviceLabel || 'Urządzenie')
    .replace(/[\\\/:*?"<>|\u0000-\u001F]/g, ' ')
    .replace(/\s+/g, ' ').trim().slice(0, 80) || 'Urządzenie';
  var created = rootFolder.createFolder(cleanLabel + suffix);
  properties.setProperty(propertyKey, created.getId());
  return created;
}

function organizeExistingDiagnostics() {
  var properties = PropertiesService.getScriptProperties();
  var folderId = String(properties.getProperty('DIAGNOSTICS_FOLDER_ID') ||
    properties.getProperty('DIAGNOSTICS_APP_FOLDER_ID') || '');
  if (!/^[A-Za-z0-9_-]{10,180}$/.test(folderId)) throw new Error('STORAGE_NOT_CONFIGURED');

  var labels = {
    'a1ebd3e5-e4b5-47c3-8039-290b20c23c19': 'Android Chrome 360x800',
    '887dc03e-6a82-4720-92a9-9e6e887478c1': 'iPhone Safari 430x932',
    '8795c5b1-97cb-4dda-b058-d5e64c90ad71': 'Android Chrome 384x832',
    '83592448-5fc1-452c-a22f-94ca7ff54789': 'Android Chrome 360x780'
  };
  var rootFolder = getDiagnosticsJsonRoot_(properties);
  var files = rootFolder.getFiles();
  var moved = 0;
  var skipped = 0;
  while (files.hasNext()) {
    var file = files.next();
    var match = String(file.getName() || '').match(/trasy-2\.0-\d+\.\d+\.\d+-([0-9a-f-]{36})-/i);
    if (!match) {
      skipped++;
      continue;
    }
    var installationId = match[1].toLowerCase();
    var deviceFolder = getDiagnosticsDeviceFolder_(
      rootFolder, properties, installationId, labels[installationId] || 'Urządzenie'
    );
    file.moveTo(deviceFolder);
    moved++;
  }
  return {moved:moved, skipped:skipped};
}

function createTestDiagnosticsSheet_(spreadsheet) {
  var sheet = spreadsheet.insertSheet('DIAGNOSTYKA TESTÓW');
  sheet.getRange(1, 1, 1, 19).setValues([[
    'BATCH_ID','SESSION_ID','INSTALACJA','WERSJA','WYSŁANO','START','KONIEC','TRASA','ZMIANA','KIERUNEK',
    'ZDARZENIA','GPS','POMINIĘTE','BŁĘDY','PIERWSZE_ZDARZENIE','OSTATNIE_ZDARZENIE','CZĘŚĆ','PLIK_JSON','URZĄDZENIE'
  ]]);
  sheet.setFrozenRows(1);
  return sheet;
}
